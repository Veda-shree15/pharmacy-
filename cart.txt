<script>
// Update subtotal and total
function updateTotals() {
  let total = 0;
  document.querySelectorAll("#cart-list li").forEach(li => {
    const qty = parseInt(li.querySelector(".quantity").innerText);
    const price = parseFloat(li.querySelector(".unit-price").innerText);
    const subtotal = qty * price;
    li.querySelector(".subtotal").innerText = subtotal.toFixed(2);
    total += subtotal;
  });
  document.getElementById("total-amount").innerText = total.toFixed(2);
}

updateTotals();

// Helper to handle login-required responses
function handleResponse(res) {
  if (res.status === 401) {
    alert("Please login to modify the cart!");
    window.location.href = "/login"; // Or trigger login modal
    throw "login_required";
  }
  return res.json();
}

// Increment buttons
document.querySelectorAll(".increment-btn").forEach(btn => {
  btn.addEventListener("click", function() {
    const li = this.closest("li");
    const product_id = li.dataset.id;
    fetch("/add_to_cart", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({product_id})
    })
    .then(handleResponse)
    .then(data => {
      li.querySelector(".quantity").innerText = data.quantity;
      document.getElementById("cart-count").innerText = data.cart_count;
      updateTotals();
    })
    .catch(err => { if (err !== "login_required") console.error(err); });
  });
});

// Decrement buttons
document.querySelectorAll(".decrement-btn").forEach(btn => {
  btn.addEventListener("click", function() {
    const li = this.closest("li");
    const product_id = li.dataset.id;
    fetch("/decrement_cart", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({product_id})
    })
    .then(handleResponse)
    .then(data => {
      if (data.quantity === 0) li.remove();
      else li.querySelector(".quantity").innerText = data.quantity;
      document.getElementById("cart-count").innerText = data.cart_count;
      updateTotals();
    })
    .catch(err => { if (err !== "login_required") console.error(err); });
  });
});

// Remove buttons
document.querySelectorAll(".remove-from-cart").forEach(btn => {
  btn.addEventListener("click", function() {
    const li = this.closest("li");
    const product_id = li.dataset.id;
    fetch("/remove_from_cart", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({product_id})
    })
    .then(handleResponse)
    .then(data => {
      li.remove();
      document.getElementById("cart-count").innerText = data.cart_count;
      updateTotals();
    })
    .catch(err => { if (err !== "login_required") console.error(err); });
  });
});

// Show receipt
// Show receipt and save order to backend
function showReceipt(method) {
  const total = document.getElementById("total-amount").innerText;

  // Send cart to backend
  fetch("/place_order", {method:"POST"})
    .then(res => res.json())
    .then(data => {
        if(data.success){
            // Show receipt
            document.getElementById("receipt-message").innerText = "Payment Method: " + method;
            document.getElementById("receipt-amount").innerText = total;

            document.getElementById("pay-cash").style.display = "none";
            document.getElementById("pay-upi").style.display = "none";
            document.getElementById("pay-card").style.display = "none";
            document.getElementById("card-payment").style.display = "none";
            document.getElementById("upi-qr").style.display = "none";

            document.getElementById("payment-receipt").style.display = "block";
        } else {
            alert("Error placing order: " + data.message);
        }
    });
}


// Cash payment
document.getElementById("pay-cash")?.addEventListener("click", () => showReceipt("Cash"));

// UPI payment
document.getElementById("pay-upi")?.addEventListener("click", () => {
  document.getElementById("card-payment").style.display = "none";
  document.getElementById("upi-qr").style.display = "block";
});

// Card payment
document.getElementById("pay-card")?.addEventListener("click", () => {
  document.getElementById("upi-qr").style.display = "none";
  document.getElementById("card-payment").style.display = "block";
  document.getElementById("card-total").innerText = document.getElementById("total-amount").innerText;
});

// Card form submit
document.getElementById("card-form")?.addEventListener("submit", function(e){
  e.preventDefault();
  const cardNumber = document.getElementById("card-number").value.trim();
  const cardHolder = document.getElementById("card-holder").value.trim();
  const expiry = document.getElementById("expiry").value.trim();
  const cvv = document.getElementById("cvv").value.trim();

  if(!/^\d{16}$/.test(cardNumber)){ alert("Card number must be 16 digits!"); return; }
  if(cardHolder.length < 3){ alert("Enter valid card holder name!"); return; }
  if(!/^\d{2}\/\d{2}$/.test(expiry)){ alert("Expiry must be in MM/YY format!"); return; }
  if(!/^\d{3}$/.test(cvv)){ alert("CVV must be 3 digits!"); return; }

  showReceipt("Card");
});

// Close receipt
document.getElementById("close-receipt")?.addEventListener("click", () => {
  fetch("/clear_cart", {method:"POST"}).then(res => location.reload());
});
</script>
